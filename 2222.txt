void CatiaAutoCheckSPQ_APICmd::SPQ_API_SortPointsAlongDirection(const CATMathVector& direction, std::vector<CATMathPoint>& ioPoints) {
	std::vector<std::pair<double, CATMathPoint>> pointsProject;
	for (const CATMathPoint& point : ioPoints) {
		CATMathVector pointVec = point - CATMathPoint(0.0, 0.0, 0.0);
		double projectValue = pointVec * direction;
		pointsProject.emplace_back(projectValue, point);
	}

	std::sort(pointsProject.begin(), pointsProject.end(), [](const auto& point1, const auto& point2) {
		return 	point1.first < point2.first;
	});

	for (int i = 0; i < pointsProject.size(); i++) {
		auto& projectPoint = pointsProject[i];
		ioPoints[i] = projectPoint.second;
	}
}


HRESULT MDSUtility::Get3DRep(CATBaseUnknown* iObject, CAT3DRep** oRep)
{
	HRESULT rc = E_FAIL;

	if (nullptr == oRep) {
		return rc;
	}

	CATVisManager* pVisManager = CATVisManager::GetVisManager();

	CATViewpoint* pViewpoint = nullptr;

	CATFrmLayout* pLayout = CATFrmLayout::GetCurrentLayout();
	if (nullptr != pLayout) {
		CATFrmWindow* pWindow = pLayout->GetCurrentWindow();
		if (nullptr != pWindow) {
			CATViewer* pViewer = pWindow->GetViewer();

			if (nullptr != pViewer) {
				CAT3DViewpoint& Main3DViewpoint = pViewer->GetMain3DViewpoint();
				pViewpoint = (CATViewpoint*)(&Main3DViewpoint);
			}
		}
	}

	CATFrmEditor* pEditor = CATFrmEditor::GetCurrentEditor();;
	CATBaseUnknown* pUIActiveObject = nullptr;
	if (nullptr != pEditor) {
		CATPathElement path = pEditor->GetUIActiveObject();
		if (0 != path.GetSize()) {
			pUIActiveObject = path[path.GetSize() - 1];
		}
	}

	if ((nullptr != pVisManager) && (nullptr != pViewpoint) && (nullptr != pUIActiveObject)) {
		// Create a path with the Root object
		CATPathElement Path;

		Path.AddChildElement(pUIActiveObject);
		if (nullptr != iObject) {
			Path.AddChildElement(iObject);
		}

		CATRepPath RepPath;

		// 代表的产生
		rc = pVisManager->GenerateRepPathFromPathElement(Path, pViewpoint, RepPath);
		if (RepPath.Size() >= 1) {
			*oRep = (CAT3DRep*)RepPath[RepPath.Size() - 1];
			if (nullptr != *oRep) {
				rc = S_OK;
			}
		}
	}
	return rc;
}
